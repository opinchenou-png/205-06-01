function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), data.seat, data.name, data.score, data.accuracy]);
  return ContentService.createTextOutput(JSON.stringify({"status":"success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  var rows = data.slice(1);
  rows.sort(function(a, b) { return Number(b[3]) - Number(a[3]); });
  var top5 = rows.slice(0, 5).map(function(r) {
    return { seat: String(r[1]), name: String(r[2]), score: Number(r[3]), accuracy: String(r[4]) };
  });
  return ContentService.createTextOutput(JSON.stringify(top5)).setMimeType(ContentService.MimeType.JSON);
}
